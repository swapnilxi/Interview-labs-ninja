1. Why U-Net for crack detection?

Cracks in concrete, asphalt, or metal are thin, elongated, and high-contrast structures. Semantic segmentation assigns a label to each pixel — perfect for this task. U-Net (Ronneberger et al., 2015) remains a gold standard for biomedical and industrial segmentation because its skip connections preserve fine spatial detail while the bottleneck captures global context.

U-Net architecture: contracting path (encoder) + expansive path (decoder) with skip connections.

2. Training a U-Net in PyTorch

We’ll use a compact U-Net implementation. The loss combines Binary Cross-Entropy (BCE) and Dice loss — a common choice for thin structures.

Model definition (simplified)

import torch
import torch.nn as nn

class DoubleConv(nn.Module):
    def __init__(self, in_c, out_c):
        super().__init__()
        self.conv = nn.Sequential(
            nn.Conv2d(in_c, out_c, 3, padding=1),
            nn.BatchNorm2d(out_c),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_c, out_c, 3, padding=1),
            nn.BatchNorm2d(out_c),
            nn.ReLU(inplace=True)
        )
    def forward(self, x):
        return self.conv(x)

class UNet(nn.Module):
    def __init__(self, in_channels=3, out_channels=1):
        super().__init__()
        # encoder
        self.down1 = DoubleConv(in_channels, 64)
        self.down2 = DoubleConv(64, 128)
        self.down3 = DoubleConv(128, 256)
        self.down4 = DoubleConv(256, 512)
        self.pool = nn.MaxPool2d(2)
        # bottleneck
        self.bottleneck = DoubleConv(512, 1024)
        # decoder
        self.up4 = nn.ConvTranspose2d(1024, 512, 2, stride=2)
        self.up3 = nn.ConvTranspose2d(512, 256, 2, stride=2)
        self.up2 = nn.ConvTranspose2d(256, 128, 2, stride=2)
        self.up1 = nn.ConvTranspose2d(128, 64, 2, stride=2)
        self.conv_up4 = DoubleConv(1024, 512)
        self.conv_up3 = DoubleConv(512, 256)
        self.conv_up2 = DoubleConv(256, 128)
        self.conv_up1 = DoubleConv(128, 64)
        self.out = nn.Conv2d(64, out_channels, 1)

    def forward(self, x):
        # encoder
        d1 = self.down1(x)
        d2 = self.down2(self.pool(d1))
        d3 = self.down3(self.pool(d2))
        d4 = self.down4(self.pool(d3))
        # bottleneck
        b = self.bottleneck(self.pool(d4))
        # decoder + skip connections
        u4 = self.up4(b)
        u4 = torch.cat([u4, d4], dim=1)
        u4 = self.conv_up4(u4)
        u3 = self.up3(u4)
        u3 = torch.cat([u3, d3], dim=1)
        u3 = self.conv_up3(u3)
        u2 = self.up2(u3)
        u2 = torch.cat([u2, d2], dim=1)
        u2 = self.conv_up2(u2)
        u1 = self.up1(u2)
        u1 = torch.cat([u1, d1], dim=1)
        u1 = self.conv_up1(u1)
        return torch.sigmoid(self.out(u1))

Training loop (BCE + Dice)

def dice_loss(pred, target, smooth=1e-6):
    pred = pred.contiguous().view(-1)
    target = target.contiguous().view(-1)
    intersection = (pred * target).sum()
    return 1 - ((2. * intersection + smooth) / (pred.sum() + target.sum() + smooth))

app.py (Gradio interface)

import gradio as gr
import torch
import numpy as np
from PIL import Image
import torchvision.transforms as T

# load model (same UNet class as above)
model = UNet(in_channels=3, out_channels=1)
model.load_state_dict(torch.load("model_unet.pth", map_location="cpu"))
model.eval()

transform = T.Compose([
    T.Resize((256, 256)),
    T.ToTensor(),
])

def predict(image):
    img_tensor = transform(image).unsqueeze(0)
    with torch.no_grad():
        mask = model(img_tensor).squeeze().cpu().numpy()
    mask = (mask > 0.5).astype(np.uint8) * 255
    return Image.fromarray(mask)

gr.Interface(
    fn=predict,
    inputs=gr.Image(type="pil"),
    outputs=gr.Image(type="pil"),
    title="Crack Detection with U-Net",
    description="Upload a surface image to detect cracks."
).launch()
