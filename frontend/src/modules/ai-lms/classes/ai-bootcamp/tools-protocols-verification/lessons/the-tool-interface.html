The Anatomy of a Tool

Every AI tool consists of three distinct parts: a Name, a Description that guides LLM intent, and a JSON Schema declaring argument parameters and types.

from pydantic import BaseModel, Field
from typing import Literal

class WeatherQuery(BaseModel):
    location: str = Field(description="The city and state, e.g. San Francisco, CA")
    units: Literal["celsius", "fahrenheit"] = Field(default="celsius", description="Temperature unit")

# Convert to OpenAI-compatible Tool Schema
tool_def = {
    "type": "function",
    "function": {
        "name": "get_current_weather",
        "description": "Get current weather conditions for a given location",
        "parameters": WeatherQuery.model_json_schema()
    }
}
print(tool_def)
